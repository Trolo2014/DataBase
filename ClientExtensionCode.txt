                function formatTimeLeft(totalSec) {
                    const days = Math.floor(totalSec / 86400);
                    let rem = totalSec % 86400;
                    const hours = Math.floor(rem / 3600);
                    rem %= 3600;
                    const minutes = Math.floor(rem / 60);
                    const seconds = rem % 60;
                    return `${days}D ${hours}H ${minutes}M ${seconds}S`;
                }

                async function checkLiveStatus() {
                    return new Promise((resolve) => {
                        if (typeof chrome === "undefined" || !chrome.runtime || !chrome.runtime.sendMessage) {
                            resolve({ level: 'yellow', message: 'PhantomScope Server Is Down' });
                            return;
                        }
                        chrome.runtime.sendMessage({ type: 'FETCH_JSON', url: 'https://phantomscope.onrender.com/api/livechannelstatus' }, (resp) => {
                            if (!resp || resp.success === false) {
                                resolve({ level: 'yellow', message: 'PhantomScope Server Is Down Or Ur Network Is Ass' });
                                return;
                            }

                            const statusCode = resp.status;
                            const okFlag = typeof resp.ok !== 'undefined' ? resp.ok : true;
                            const data = resp.data || resp;

                            if (statusCode === 403) {
                                resolve({ level: 'yellow', message: 'PhantomScope Whitelist Error' });
                            } else if (statusCode === 500 || statusCode === 503) {
                                resolve({ level: 'yellow', message: 'PhantomScope Server Is Restarting' });
                            } else if (!okFlag) {
                                resolve({ level: 'yellow', message: 'PhantomScope Server Is Down Or Ur Network Is Ass' });
                            } else {
                                try {
                                    if (data && typeof data.count === 'number' && data.count > 0) {
                                        resolve({ level: 'green', message: 'Live Is Up' });
                                    } else {
                                        resolve({ level: 'red', message: "Ping Shawn Only When #live-board Is Down | If Not Will Self Fix In Couple Minutes" });
                                    }
                                } catch (e) {
                                    resolve({ level: 'yellow', message: 'PhantomScope Server Is Down Or Ur Network Is Ass' });
                                }
                            }
                        });
                    });
                }

                function createSearchSocket(username, placeId) {
                    return new Promise((resolve, reject) => {
                        const waitForKey = setInterval(() => {
                            if (window.__Primary && window.__Primary.key) {
                                clearInterval(waitForKey);
                                const ws = new WebSocket(
                                    `wss://phantomscope.onrender.com/sniper?authkey=${encodeURIComponent(window.__Primary.key)}`
                                );
                                const startTime = Date.now();
                                ws.addEventListener("open", () => {
                                    ws.send(JSON.stringify({ username, placeId }));
                                    resolve({ ws, startTime });
                                });

                                ws.addEventListener("error", err => {
                                    reject(err);
                                });
                            }
                        }, 50);
                    });
                }


                async function wsDoSearch(autoJoin) {
                    if (searchInProgress) return false;
                    searchInProgress = true;
                    disableUIExceptAuto();
                    status.innerHTML = "🔗 Verifying Auth…";
                    joinBtn.style.display = inviteBtn.style.display = specialBtn.style.display = "none";
                    const username = input.value.trim();
                    try {
                        const { ws, startTime } = await createSearchSocket(username, placeId);
                        return await new Promise(resolve => {
                            ws.addEventListener("message", async ev => {
                                let msg;
                                try {
                                    msg = JSON.parse(ev.data);
                                } catch {
                                    console.warn("Non-JSON frame:", ev.data);
                                    return;
                                }
                                switch (msg.status) {
                                    case "update":
                                        status.innerHTML = msg.msg;
                                        break;
                                    case "done":
                                        ws.close();
                                        const secs = ((Date.now() - startTime) / 1000).toFixed(0);
                                        if (msg.result === "found") {
                                            const info = msg.data;
                                            const foundPlaceStr = String(info.placeId);
                                            const isFastMode = info.placename.includes("Fast Mode");
                                            const isMain = (foundPlaceStr === MAIN_PLACE_ID);
                                            [joinBtn, inviteBtn, specialBtn].forEach(btn => {
                                                delete btn.dataset.profileUrl;
                                                delete btn.dataset.server;
                                                delete btn.dataset.placeid;
                                                delete btn.dataset.placename;
                                                delete btn.dataset.teleportCode;
                                            });
                                            if (isFastMode && !isMain) {
                                                joinBtn.innerText = "Open Profile";
                                                joinBtn.dataset.profileUrl =
                                                    `https://www.roblox.com/users/${targetUserId}/profile?friendshipSourceType=PlayerSearch`;
                                                joinBtn.style.display = "";
                                                inviteBtn.innerText = "Invite (Profile)";
                                                inviteBtn.dataset.teleportCode = joinBtn.dataset.profileUrl;
                                                inviteBtn.style.display = "";
                                                specialBtn.innerText = "Copy Teleport Code";
                                                specialBtn.dataset.teleportCode =
                                                    `game:GetService("TeleportService"):TeleportToPlaceInstance(${info.placeId}, "${info.serverId}", game.Players.LocalPlayer)`;
                                                specialBtn.style.display = "";
                                                if (autoJoin) {
                                                    window.open(joinBtn.dataset.profileUrl, '_blank');
                                                }
                                            } else if (isMain) {
                                                joinBtn.innerText = "Join";
                                                joinBtn.dataset.server = info.serverId;
                                                joinBtn.dataset.placeid = foundPlaceStr;
                                                joinBtn.dataset.placename = info.placename;
                                                joinBtn.style.display = "";
                                                inviteBtn.innerText = "Invite";
                                                inviteBtn.dataset.teleportCode =
                                                    `https://fern.wtf/joiner?placeId=${foundPlaceStr}&gameInstanceId=${info.serverId}`;
                                                inviteBtn.style.display = "";
                                                if (autoJoin) {
                                                    chrome.runtime.sendMessage({
                                                        type: "join",
                                                        FastJoinMode: joinWithoutCloseCheckbox.checked,
                                                        placeId: foundPlaceStr,
                                                        serverId: info.serverId
                                                    });
                                                }
                                            } else {
                                                joinBtn.style.display = "none";
                                                inviteBtn.innerText = "Copy Teleport Code";
                                                inviteBtn.dataset.teleportCode =
                                                    `game:GetService("TeleportService"):TeleportToPlaceInstance(${info.placeId}, "${info.serverId}", game.Players.LocalPlayer)`;
                                                inviteBtn.style.display = "";
                                            }
                                            status.innerHTML =
                                                `⏱️ Time: ${secs}s<br>` +
                                                `🛰️ Place Name: ${info.placename}<br>` +
                                                `👤 Players: ${info.players}`;
                                            searchInProgress = false;
                                            enableUI();
                                            resolve(true);
                                        } else {
                                            status.innerHTML = msg.msg;
                                            searchInProgress = false;
                                            enableUI();
                                            resolve(false);
                                        }
                                        break;
                                    case "error":
                                        ws.close();
                                        status.innerHTML = `❌ ${msg.msg}`;
                                        searchInProgress = false;
                                        enableUI();
                                        resolve(false);
                                        break;
                                }
                            });

                            ws.addEventListener("close", () => {
                                if (searchInProgress) {
                                    searchInProgress = false;
                                    enableUI();
                                    resolve(false);
                                }
                            });

                            ws.addEventListener("error", (err) => {
                                console.warn("Sniper WS error", err);
                                if (searchInProgress) {
                                    searchInProgress = false;
                                    enableUI();
                                    resolve(false);
                                }
                            });
                        });
                    } catch (e) {
                        status.innerHTML = `❌ ${e.message || "Connection Error"}`;
                        searchInProgress = false;
                        enableUI();
                        return false;
                    }
                }

                let liveWS = null;
                let liveWSKey = null;
                let livePingInterval = null;
                let statusPollInterval = null;

                function disconnectLiveWS() {
                    if (liveWS) {
                        try { liveWS.close(); } catch {}
                        liveWS = null;
                    }

                    liveWSKey = null;

                    if (livePingInterval) {
                        clearInterval(livePingInterval);
                        livePingInterval = null;
                    }

                    if (statusPollInterval) {
                        clearInterval(statusPollInterval);
                        statusPollInterval = null;
                    }
                }

                function connectLiveWS() {
                    const currentKey = window.__Primary?.key ?? null;

                    if (liveWS && liveWSKey !== currentKey) {
                        disconnectLiveWS();
                    }

                    if (liveWS && liveWS.readyState === WebSocket.OPEN) return;

                    if (!currentKey) {
                        const waitForKey = setInterval(() => {
                            const newKey = window.__Primary?.key ?? null;
                            if (newKey) {
                                clearInterval(waitForKey);
                                connectLiveWS();
                            }
                        }, 50);
                        return;
                    }

                    liveWSKey = currentKey;
                    liveWS = new WebSocket(
                        `wss://phantomscope.onrender.com/livechannel?authkey=${encodeURIComponent(currentKey)}`
                    );

                    liveWS.addEventListener("open", () => {
                        renderStatusToElement({ level: 'green', message: 'Live WS Connected' });

                        livePingInterval = setInterval(() => {
                            if (liveWS?.readyState === WebSocket.OPEN) {
                                liveWS.send("Alive");
                            }
                        }, 1000);

                        if (!statusPollInterval) {
                            statusPollInterval = setInterval(async () => {
                                const st = await checkLiveStatus();
                                renderStatusToElement(st);
                            }, 5000);
                        }
                    });

                    liveWS.addEventListener("message", async (ev) => {
                        let data;
                        try {
                            data = JSON.parse(ev.data);
                        } catch {
                            return;
                        }

                        if (!data || typeof data !== "object") return;

                        const currentIds = new Set(Object.keys(data));

                        for (const id of Array.from(seen)) {
                            if (!currentIds.has(id)) {
                                document.getElementById(id)?.remove();
                                seen.delete(id);
                            }
                        }

                        for (const [id, info] of Object.entries(data)) {
                            if (!seen.has(id) && info) {
                                await renderResultToLivePanel(info, id);
                            }
                        }
                    });

                    liveWS.addEventListener("close", () => {
                        renderStatusToElement({ level: 'red', message: '❌ Live Connection Closed' });
                        disconnectLiveWS();

                        setTimeout(() => {
                            const panelToggle = document.getElementById('phantom-live-toggle');
                            if (panelToggle?.checked) connectLiveWS();
                        }, 2500);
                    });
                }
